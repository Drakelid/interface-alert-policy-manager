                if (this.theme === newTheme) {\n

                this.theme = newTheme;\n

\n
                fetch('http://localhost/preferences', {\n

                    },\n
                    body: JSON.stringify({\n
                        pref: 'site_style',\n
                        value: newTheme\n
                    })\n
                }).then(() => {\n
                    if (!['dark', 'light'].includes(window.siteStyle)) {\n
                        location.reload();\n

    function themeToggleComponentData() {\n
        return {\n
            theme: window.siteStylePreference,\n
            toggleTheme(newTheme) {\n
                if (this.theme === newTheme) {\n
                    return;\n
                }\n
\n
                this.theme = newTheme;\n
                window.siteStylePreference = newTheme;\n
\n
                fetch('http://localhost/preferences', {\n
                    method: 'POST',\n
                    headers: {\n
                        'Content-Type': 'application/json',\n
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').getAttribute('content')\n
                    },\n
                    body: JSON.stringify({\n
                        pref: 'site_style',\n
                        value: newTheme\n
                    })\n
                }).then(() => {\n
                    if (!['dark', 'light'].includes(window.siteStyle)) {\n
                        location.reload();\n
                    }\n
\n
                    applySiteStyle(newTheme);\n
                });\n
            }\n
        }\n

                this.open = true;\n
                this.loading = true;\n

                this.groups = [];\n
                this.flat = [];\n
                this.controllers.forEach(c => c.abort());\n
                this.controllers = [];\n
                let seq = ++this.seq;\n
                let collected = {};\n
                let pending = this.endpoints.length;\n
                this.endpoints.forEach(url => {\n
                    let controller = new AbortController();\n
                    this.controllers.push(controller);\n
                    fetch(url + '?search=' + encodeURIComponent(q), {\n
                        signal: controller.signal,\n
                        headers: { 'X-Requested-With': 'XMLHttpRequest' }\n
                    })\n
                        .then(r => r.json())\n
                        .then(data => {\n
                            if (seq !== this.seq) { return; }\n
                            (data.groups || []).forEach(g => { collected[g.type] = g; });\n
                            this.groups = this.order.filter(t => collected[t]).map(t => collected[t]);\n
                            this.flat = this.groups.flatMap(g => g.results);\n
                        })\n
                        .catch(() => {})\n
                        .finally(() => { pending--; if (seq === this.seq && pending === 0) { this.loading = false; } });\n
                });\n
            },\n
            onKey(e) {\n
                if (e.key === 'ArrowDown') { e.preventDefault(); this.move(1); }\n

                else if (e.key === 'Enter') { e.preventDefault(); this.go(); }\n

                let i = this.flat.findIndex(it => it.url === this.active);\n

                if (url) { window.location.href = url; }\n
            },\n
            close() { this.open = false; },\n
            reset() { this.controllers.forEach(c => c.abort()); this.controllers = []; this.groups = []; this.flat = []; this.open = false; this.active = ''; this.loading = false; },\n
